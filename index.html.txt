        <button onclick="setCategory('${c}')" class="px-3.5 py-1.5 rounded-full text-xs font-bold whitespace-nowrap transition ${selectedCategory === c ? 'bg-theme text-white' : 'bg-white text-gray-600 hover:bg-gray-100'}">${c}</button>
      `).join('');
    }

    function setCategory(cat) {

      renderCategories();
      renderMenuCatalog();

    function renderMenuCatalog() {
      const q = document.getElementById('search-input').value.toLowerCase();
      const grid = document.getElementById('product-grid');
      const filtered = products.filter(p => {
        const matchCat = selectedCategory === 'Semua' || p.cat === selectedCategory;
        const matchName = p.name.toLowerCase().includes(q);
        return matchCat && matchName;
      });

      grid.innerHTML = filtered.map(p => {
        const cartItem = cart.find(c => c.id === p.id);
        const qtyInCart = cartItem ? cartItem.qty : 0;
        return `
          <div onclick="addToCart(${p.id})" class="bg-white rounded-2xl border border-gray-100 shadow-sm overflow-hidden cursor-pointer active:scale-97 transition relative flex flex-col group">
            <div class="h-28 w-full bg-gray-100 relative overflow-hidden">
              <img src="${p.img || 'https://images.unsplash.com/photo-1546069901-ba9599a7e63c?w=300'}" class="w-full h-full object-cover group-hover:scale-105 transition" />
              ${qtyInCart > 0 ? `
                <div class="absolute top-2 right-2 bg-emerald-500 text-white font-black text-xs w-6 h-6 rounded-full flex items-center justify-center shadow">
                  ${qtyInCart}
                </div>
              ` : ''}
              ${p.stock <= 5 ? `
                <span class="absolute bottom-1 left-1 bg-red-600 text-white text-[9px] font-bold px-1.5 py-0.5 rounded">
                  Sisa ${p.stock}
                </span>
              ` : ''}
            </div>
            <div class="p-2.5 flex-1 flex flex-col justify-between">
              <h4 class="font-bold text-gray-800 text-xs line-clamp-1">${p.name}</h4>
              <p class="font-black text-theme text-xs mt-1">Rp ${p.sell.toLocaleString('id-ID')}</p>
            </div>
          </div>
        `;
      }).join('');
    }

    // --- KERANJANG BELANJA ---
    function addToCart(id) {
      const prod = products.find(p => p.id === id);
      if (prod.stock <= 0) return alert('Stok produk ini habis!');
      const exist = cart.find(c => c.id === id);
      if (exist) {
        if (exist.qty + 1 > prod.stock) return alert('Jumlah melebihi stok yang tersedia!');
        exist.qty += 1;
      } else {
        cart.push({ ...prod, qty: 1 });
      }
      renderCart();
      renderMenuCatalog();
    }

    function updateCartQty(id, delta) {
      const item = cart.find(c => c.id === id);
      const prod = products.find(p => p.id === id);
      if (item) {
        item.qty += delta;
        if (item.qty > prod.stock) item.qty = prod.stock;
        if (item.qty <= 0) cart = cart.filter(c => c.id !== id);
      }
      renderCart();
      renderMenuCatalog();
    }

    function clearCart() {
      cart = [];
      renderCart();
      renderMenuCatalog();
    }

    function renderCart() {
      const list = document.getElementById('cart-item-list');
      const badge = document.getElementById('mobile-cart-badge');
      let total = 0;

      list.innerHTML = cart.map(item => {
        const sub = item.qty * item.sell;
        total += sub;
        return `
          <div class="py-2 flex justify-between items-center text-xs">
            <div class="flex-1 pr-2">
              <p class="font-bold text-gray-800">${item.name}</p>
              <p class="text-[10px] text-gray-400">Rp ${item.sell.toLocaleString('id-ID')}</p>
            </div>
            <div class="flex items-center gap-1.5">
              <button onclick="updateCartQty(${item.id}, -1)" class="w-5 h-5 bg-gray-100 rounded text-gray-700 font-bold">-</button>
              <span class="font-bold text-xs w-4 text-center">${item.qty}</span>
              <button onclick="updateCartQty(${item.id}, 1)" class="w-5 h-5 bg-gray-100 rounded text-gray-700 font-bold">+</button>
            </div>
          </div>
        `;
      }).join('');

      document.getElementById('cart-subtotal').innerText = `Rp ${total.toLocaleString('id-ID')}`;
      document.getElementById('cart-total').innerText = `Rp ${total.toLocaleString('id-ID')}`;
      badge.innerText = cart.reduce((acc, c) => acc + c.qty, 0);
    }

    // --- CHECKOUT & STRUK ---
    function openPaymentModal() {
      if (cart.length === 0) return alert('Keranjang belanja masih kosong!');
      const total = cart.reduce((acc, c) => acc + (c.qty * c.sell), 0);
      document.getElementById('pay-total-display').innerText = `Rp ${total.toLocaleString('id-ID')}`;
      document.getElementById('pay-cash-input').value = total;
      calcChange();
      document.getElementById('payment-modal').classList.remove('hidden');
    }

    function closePaymentModal() {
      document.getElementById('payment-modal').classList.add('hidden');
    }

    function setFastCash(val) {
      const total = cart.reduce((acc, c) => acc + (c.qty * c.sell), 0);
      document.getElementById('pay-cash-input').value = val === 'pas' ? total : val;
      calcChange();
    }

    function calcChange() {
      const total = cart.reduce((acc, c) => acc + (c.qty * c.sell), 0);
      const pay = Number(document.getElementById('pay-cash-input').value) || 0;
      const change = pay - total;
      document.getElementById('pay-change-display').innerText = `Rp ${Math.max(0, change).toLocaleString('id-ID')}`;
    }

    function processTransaction() {
      const total = cart.reduce((acc, c) => acc + (c.qty * c.sell), 0);
      const pay = Number(document.getElementById('pay-cash-input').value) || 0;
      if (pay < total) return alert('Uang pembayaran masih kurang!');

      const inv = 'TRX-' + Date.now().toString().slice(-6);
      const change = pay - total;
      const method = document.getElementById('pay-method-select').value;
      const timeStr = new Date().toLocaleString('id-ID');

      // Kurangi stok
      cart.forEach(c => {
        const prod = products.find(p => p.id === c.id);
        if (prod) prod.stock -= c.qty;
      });
      localStorage.setItem('qpos_products', JSON.stringify(products));

      // Simpan riwayat
      const trx = {
        invoice: inv,
        date: timeStr,
        items: [...cart],
        total,
        pay,
        change,
        method
      };
      transactions.unshift(trx);
      localStorage.setItem('qpos_transactions', JSON.stringify(transactions));

      printThermalReceipt(trx);
      clearCart();
      closePaymentModal();
      renderMenuCatalog();
      renderMenuTable();
      updateReports();
    }

    function printThermalReceipt(trx) {
      document.getElementById('rcpt-store').innerText = store.name;
      document.getElementById('rcpt-address').innerText = store.address;
      document.getElementById('rcpt-phone').innerText = store.phone;
      document.getElementById('rcpt-inv').innerText = trx.invoice;
      document.getElementById('rcpt-time').innerText = trx.date;

      document.getElementById('rcpt-items-list').innerHTML = trx.items.map(i => `
        <div style="display: flex; justify-content: space-between; margin-bottom: 2px;">
          <span>${i.name} x${i.qty}</span>
          <span>Rp ${(i.qty * i.sell).toLocaleString('id-ID')}</span>
        </div>
      `).join('');

      document.getElementById('rcpt-total').innerText = `Rp ${trx.total.toLocaleString('id-ID')}`;
      document.getElementById('rcpt-pay').innerText = `Rp ${trx.pay.toLocaleString('id-ID')}`;
      document.getElementById('rcpt-change').innerText = `Rp ${trx.change.toLocaleString('id-ID')}`;
      document.getElementById('rcpt-footer').innerText = store.footer;

      window.print();
    }

    function handleProductSubmit(e) {
      e.preventDefault();
      const id = document.getElementById('prod-id').value;
      const name = document.getElementById('prod-name').value;
      const cat = document.getElementById('prod-cat').value;
      const cost = Number(document.getElementById('prod-cost').value);
      const sell = Number(document.getElementById('prod-sell').value);
      const stock = Number(document.getElementById('prod-stock').value);
      const img = document.getElementById('prod-img-data').value;

      if (id) {
        const idx = products.findIndex(p => p.id == id);
        if (idx !== -1) {
          products[idx] = { ...products[idx], name, cat, cost, sell, stock, img: img || products[idx].img };
        }
      } else {
        products.push({ id: Date.now(), name, cat, cost, sell, stock, img: img || 'https://images.unsplash.com/photo-1546069901-ba9599a7e63c?w=300' });
      }

      localStorage.setItem('qpos_products', JSON.stringify(products));
      resetProductForm();
      renderMenuCatalog();
      renderMenuTable();
      alert('Menu berhasil disimpan!');
    }

    // --- LAPORAN & EXPORT EXCEL / PDF ---
    function updateReports() {
      let totalOmzet = 0;
      let totalProfit = 0;

      transactions.forEach(t => {
        totalOmzet += t.total;
        t.items.forEach(i => {
          totalProfit += (i.sell - (i.cost || 0)) * i.qty;
        });
      });

      const count = transactions.length;
      const avg = count > 0 ? totalOmzet / count : 0;

      document.getElementById('kpi-omzet').innerText = `Rp ${totalOmzet.toLocaleString('id-ID')}`;
      document.getElementById('kpi-profit').innerText = `Rp ${totalProfit.toLocaleString('id-ID')}`;
      document.getElementById('kpi-count').innerText = count;
      document.getElementById('kpi-avg').innerText = `Rp ${Math.round(avg).toLocaleString('id-ID')}`;

      const list = document.getElementById('reports-transaction-list');
      if (transactions.length === 0) {
        list.innerHTML = '<p class="text-xs text-gray-400">Belum ada transaksi.</p>';
        return;
      }
      list.innerHTML = transactions.map(t => `
        <div class="p-3 border rounded-xl flex justify-between items-center text-xs bg-gray-50/50">
          <div>
            <p class="font-bold text-gray-800">${t.invoice}</p>
            <p class="text-[10px] text-gray-400">${t.date} • ${t.method}</p>
            <p class="text-[11px] text-gray-600 mt-1">${t.items.map(i => `${i.name} (x${i.qty})`).join(', ')}</p>
          </div>
          <div class="text-right">
            <p class="font-black text-theme text-sm">Rp ${t.total.toLocaleString('id-ID')}</p>
          </div>
        </div>
      `).join('');
    }

    function exportToExcel() {
      if (transactions.length === 0) return alert('Belum ada transaksi untuk diekspor!');

      let csvContent = "\uFEFFNo Invoice,Tanggal,Item Terjual,Total Omzet,Metode Bayar\n";
      transactions.forEach(t => {
        const itemSummary = t.items.map(i => `${i.name} x${i.qty}`).join('; ');
        csvContent += `"${t.invoice}","${t.date}","${itemSummary}","${t.total}","${t.method}"\n`;
      });

      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.setAttribute("href", url);
      link.setAttribute("download", `Laporan_Penjualan_${Date.now()}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }

      window.print();
    }
